package org.firstinspires.ftc.teamcode;

import com.qualcomm.robotcore.eventloop.opmode.OpMode;
import com.qualcomm.robotcore.eventloop.opmode.TeleOp;
import com.qualcomm.robotcore.hardware.DcMotor;
import com.qualcomm.robotcore.hardware.DistanceSensor;
import com.qualcomm.hardware.limelightvision.LLResult;
import com.qualcomm.hardware.limelightvision.LLResultTypes;
import com.qualcomm.hardware.limelightvision.LLStatus;
import com.qualcomm.hardware.limelightvision.Limelight3A;


import org.firstinspires.ftc.robotcore.external.navigation.DistanceUnit;

import java.util.List;

@TeleOp
public class DistanceShooter_PowerChange extends OpMode {

   private DcMotor Shooter;
   private Limelight3A limelight;


   @Override
   public void init() {
       Shooter = hardwareMap.get(DcMotor.class, "Shooter");
       limelight = hardwareMap.get(Limelight3A.class, "limelight");
       limelight.pipelineSwitch(0);
       limelight.setPollRateHz(100); // This sets how often we ask Limelight for data (100 times per second)
       limelight.start(); // This tells Limelight to start looking!

   }

   @Override
   public void loop() {
       LLResult result = limelight.getLatestResult();

       if (result.isValid()) {
           List<LLResultTypes.FiducialResult> tags = result.getFiducialResults();

           if (!tags.isEmpty()) {
               LLResultTypes.FiducialResult tag = tags.get(0);

               double distance = tag.getTargetPoseRobotSpace().getPosition().z; // meters
               double power = Math.min(1, Math.abs(distance * 0.28));

               telemetry.addData("Power", power);
               if (power > 1.0) {
                   power = 1;
               }
               if (gamepad1.a) {
                   Shooter.setPower(power);
               } else {
                   Shooter.setPower(0);
               }
           } else {
               telemetry.addData("Status", "No tags detected");
               Shooter.setPower(0);
           }
       } else {
           telemetry.addData("Status", "No valid result");
           Shooter.setPower(0);
       }
   }
}
